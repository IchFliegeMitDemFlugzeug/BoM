Option Explicit                                                                    ' Требуем явного объявления всех переменных (чтобы ловить ошибки сразу)

' -----------------------------                                                    ' Блок констант: номера колонок таблицы
Const COL_A_FIND = 1                                                              ' Колонка A: "Найти (ЭЧ)"
Const COL_G1 = 2                                                                  ' Колонка B: "G1"
Const COL_G8 = 9                                                                  ' Колонка I: "G8"
Const COL_J_DEF = 10                                                              ' Колонка J: "Наименование (ЭМ)"
Const COL_K_PARENT = 11                                                           ' Колонка K: "Входимость"
Const COL_L_QTY = 12                                                              ' Колонка L: "Кол-во на 1 сборку"
Const COL_M_CODE = 13                                                             ' Колонка M: "код"
Const COL_N_TYPE = 14                                                             ' Колонка N: "Композит/металл"
Const COL_O_MAT = 15                                                              ' Колонка O: "Материал"
Const COL_P_MASS_EACH = 16                                                        ' Колонка P: "Масса 1 шт. (кг)"
Const COL_Q_MASS_TOTAL = 17                                                       ' Колонка Q: "Суммарная масса (кг)"

Dim payloadPath                                                                    ' Путь к TSV payload (передаёт CATScript)
Dim outXlsx                                                                        ' Путь к итоговому Excel .xlsx
Dim startLevel                                                                     ' Стартовый G-уровень (1..8), который передаёт CATScript

' -----------------------------                                                    ' Проверяем аргументы запуска (их должен передать CATScript)
If WScript.Arguments.Count < 3 Then                                               ' Если аргументов меньше 3
    WScript.Echo "Usage: ExcelBuilder.vbs payload.tsv output.xlsx startLevel(1..8)" ' Печатаем подсказку
    WScript.Quit 1                                                                ' Завершаем работу с кодом ошибки
End If                                                                            ' Конец проверки количества аргументов

payloadPath = WScript.Arguments(0)                                                ' Берём 1-й аргумент: путь к payload
outXlsx = WScript.Arguments(1)                                                    ' Берём 2-й аргумент: путь куда сохранить .xlsx
startLevel = CLng(WScript.Arguments(2))                                           ' Берём 3-й аргумент и приводим к целому (1..8)

If startLevel < 1 Or startLevel > 8 Then                                          ' Если уровень вне диапазона
    startLevel = 2                                                                ' Делаем безопасный фолбэк: G2
End If                                                                            ' Конец проверки стартового уровня

' -----------------------------                                                    ' Читаем payload TSV в Unicode (UTF-16 LE)
Dim text                                                                           ' Переменная для всего текста payload
text = ReadUnicodeFile(payloadPath)                                               ' Читаем Unicode-файл через ADODB.Stream

If Len(Trim(text)) = 0 Then                                                       ' Если payload пустой или не прочитан
    WScript.Echo "Payload is empty: " & payloadPath                               ' Сообщаем в консоль WSH
    WScript.Quit 2                                                                ' Завершаем работу с ошибкой
End If                                                                            ' Конец проверки payload

Dim lines                                                                          ' Массив строк payload
lines = Split(text, vbCrLf)                                                       ' Делим текст на строки по CRLF

' -----------------------------                                                    ' Поднимаем Excel через COM
Dim xl                                                                             ' Объект Excel.Application
Set xl = Nothing                                                                  ' Инициализация (на случай ошибки создания)

On Error Resume Next                                                              ' Включаем перехват ошибок COM
Set xl = CreateObject("Excel.Application")                                        ' Создаём экземпляр Excel
On Error GoTo 0                                                                   ' Возвращаем стандартную обработку ошибок

If xl Is Nothing Then                                                             ' Если Excel не создался (нет Excel/COM)
    WScript.Echo "Excel COM is not available."                                    ' Сообщаем
    WScript.Quit 3                                                                ' Завершаем
End If                                                                            ' Конец проверки Excel

xl.Visible = False                                                                ' Не показываем Excel пользователю
xl.DisplayAlerts = False                                                          ' Отключаем диалоги "Сохранить/Перезаписать"

Dim wb                                                                             ' Workbook (книга Excel)
Set wb = xl.Workbooks.Add()                                                       ' Создаём новую книгу

Dim ws                                                                             ' Worksheet (лист Excel)
Set ws = wb.Worksheets(1)                                                         ' Берём первый лист книги

On Error Resume Next                                                              ' На всякий случай перехват (имя может конфликтовать)
ws.Name = "BOM"                                                                   ' Называем лист латиницей (чтобы не зависеть от локали)
On Error GoTo 0                                                                   ' Возвращаем стандартную обработку ошибок

' -----------------------------                                                    ' Пишем шапку (строка 1)
Call WriteHeader(ws)                                                              ' Заполняем заголовки A..Q

Dim row                                                                            ' Текущая строка Excel для записи данных
row = 2                                                                            ' Данные начинаем со 2-й строки (1-я — шапка)

' -----------------------------                                                    ' Стек для сборок: чтобы группировать потомков каждой сборки
Dim stackDepth()                                                                  ' Массив глубин сборок (Depth)
Dim stackRow()                                                                    ' Массив строк Excel, где стоит строка сборки (summary)
Dim sp                                                                            ' Указатель вершины стека
ReDim stackDepth(200)                                                             ' Резервируем место под 200 уровней вложенности
ReDim stackRow(200)                                                               ' Резервируем место под 200 уровней вложенности
sp = -1                                                                           ' Стек пуст (вершины нет)

' -----------------------------                                                    ' Основной проход по строкам payload
Dim i                                                                             ' Счётчик строк payload
For i = 1 To UBound(lines)                                                        ' Строку 0 пропускаем (это заголовок TSV)

    Dim ln                                                                         ' Текущая строка payload
    ln = lines(i)                                                                 ' Берём строку

    If Trim(ln) <> "" Then                                                        ' Если строка не пустая

        Dim f                                                                      ' Массив полей строки
        f = Split(ln, vbTab)                                                       ' Делим по табуляции

        Dim depth                                                                  ' Глубина в дереве сборки (0,1,2...)
        depth = ParseLong(SafeField(f, 0))                                         ' Парсим Depth безопасно в целое

        Dim kind                                                                   ' Тип строки: ASM/PART/FAST/PKI/...
        kind = SafeField(f, 1)                                                     ' Берём Kind

        Dim label                                                                  ' Что печатать в уровневую колонку (G*)
        label = SafeField(f, 2)                                                    ' Берём Label

        Dim partNumber                                                             ' Обозначение/PartNumber (может быть пустым у крепежа/ПКИ)
        partNumber = SafeField(f, 3)                                               ' Берём PartNumber

        Dim defText                                                                ' Definition (идёт в J)
        defText = SafeField(f, 4)                                                  ' Берём Definition

        Dim parentPN                                                               ' Родительская сборка (идёт в K "Входимость")
        parentPN = SafeField(f, 5)                                                 ' Берём ParentPN

        Dim qty                                                                    ' Количество (идёт в L)
        qty = ParseLong(SafeField(f, 6))                                           ' Парсим Qty

        Dim codeM                                                                  ' Код (идёт в M)
        codeM = SafeField(f, 7)                                                    ' Берём CodeM

        Dim typeN                                                                  ' Тип (идёт в N)
        typeN = SafeField(f, 8)                                                    ' Берём TypeN

        Dim matO                                                                   ' Материал (идёт в O)
        matO = SafeField(f, 9)                                                     ' Берём MaterialO

        Dim massEach                                                               ' Масса 1 шт (идёт в P)
        massEach = ParseLocaleNumber(SafeField(f, 10))                             ' Парсим MassEachKg

        Dim massTotal                                                              ' Суммарная масса (идёт в Q)
        massTotal = ParseLocaleNumber(SafeField(f, 11))                            ' Парсим MassTotalKg

        ' -----------------------------                                            ' Закрываем группы сборок, если мы "поднялись вверх" по дереву
        ' ВАЖНО: endRow = row-1, потому что текущую строку мы ещё не записали       ' Пояснение логики
        Call CloseAsmGroups(ws, stackDepth, stackRow, sp, depth, row - 1)          ' Закрываем все сборки глубже текущей глубины

        ' -----------------------------                                            ' Записываем строку данных в Excel
        Call WriteExcelRow(ws, row, depth, startLevel, label, defText, parentPN, qty, codeM, typeN, matO, massEach, massTotal) ' Заполняем ячейки строки

        ' -----------------------------                                            ' Если строка — это сборка, то добавляем её в стек
        If UCase(kind) = "ASM" Then                                               ' Если Kind = ASM (подсборка/сборка)
            sp = sp + 1                                                           ' Сдвигаем вершину стека вверх
            stackDepth(sp) = depth                                                ' Запоминаем глубину сборки
            stackRow(sp) = row                                                    ' Запоминаем строку Excel, где стоит сборка (summary row)
        End If                                                                    ' Конец проверки "это сборка?"

        row = row + 1                                                             ' Переходим к следующей строке Excel
    End If                                                                        ' Конец проверки пустой строки
Next                                                                              ' Следующая строка payload

' -----------------------------                                                    ' В конце файла закрываем ВСЕ оставшиеся сборки в стеке
Call CloseAsmGroups(ws, stackDepth, stackRow, sp, -1, row - 1)                    ' Закрываем группы до конца (newDepth=-1 => закрыть всё)

Dim lastRow                                                                       ' Последняя заполненная строка Excel
lastRow = row - 1                                                                 ' Вычисляем последнюю строку данных

' -----------------------------                                                    ' Включаем фильтр по шапке (строка 1)
ws.Range(ws.Cells(1, 1), ws.Cells(1, COL_Q_MASS_TOTAL)).AutoFilter                ' AutoFilter на диапазон A1:Q1

' -----------------------------                                                    ' Форматируем лист, настраиваем outline и группировку столбцов
Call FormatSheet(ws, lastRow)                                                     ' Ширины, границы, шапка, outline, freeze panes, группы

' -----------------------------                                                    ' Сохраняем и закрываем Excel
wb.SaveAs outXlsx                                                                 ' Сохраняем книгу по заданному пути
wb.Close False                                                                    ' Закрываем книгу без сохранения повторно
xl.Quit                                                                           ' Закрываем Excel

Set ws = Nothing                                                                  ' Освобождаем ссылку на лист
Set wb = Nothing                                                                  ' Освобождаем ссылку на книгу
Set xl = Nothing                                                                  ' Освобождаем ссылку на Excel

WScript.Quit 0                                                                    ' Успешное завершение


' ============================================================================= ' Раздел: ГРУППИРОВКИ ПО СБОРКАМ (ROW OUTLINE)
' =============================================================================
Sub CloseAsmGroups(ws, ByRef stackDepth, ByRef stackRow, ByRef sp, ByVal newDepth, ByVal endRow) ' Процедура закрытия (группировки) сборок, которые закончились

    ' Логика: если следующий элемент (newDepth) пришёл на уровень ТОТ ЖЕ или ЛЕВЕЕ (меньше/равно),
    ' значит текущая подсборка на вершине стека уже завершилась и её потомков надо сгруппировать.
    ' Поэтому условие закрытия: stackDepth(sp) >= newDepth (а не только >).                                  ' Поясняем ключевое отличие

    Do While sp >= 0                                                                                          ' Пока в стеке есть открытые сборки (sp — индекс вершины стека)

        If CLng(stackDepth(sp)) >= CLng(newDepth) Then                                                        ' Если глубина сборки на вершине стека >= глубины следующего элемента, сборка закончилась

            Dim asmRow                                                                                        ' Переменная для строки Excel, где записана сама сборка (summary row)
            asmRow = CLng(stackRow(sp))                                                                       ' Берём строку сборки из стека и приводим к целому числу

            ' Потомки сборки находятся СТРОГО ниже строки сборки, начиная с asmRow+1 и до последней строки потомков (endRow).
            ' ВАЖНО: мы группируем ТОЛЬКО потомков, а не строку сборки — тогда кнопка +/- будет напротив сборки (сверху).       ' Пояснение
            If endRow >= asmRow + 1 Then                                                                      ' Если у сборки есть хотя бы одна строка потомков (есть что группировать)
                ws.Rows(CStr(asmRow + 1) & ":" & CStr(endRow)).Group                                          ' Группируем диапазон потомков сборки (Excel создаст +/- у строки сборки)
            End If                                                                                            ' Конец проверки "есть ли потомки"

            sp = sp - 1                                                                                       ' Удаляем (закрываем) эту сборку со стека, т.к. её ветка полностью обработана

        Else                                                                                                  ' Иначе: вершина стека глубже не требует закрытия, т.к. мы всё ещё внутри этой сборки
            Exit Do                                                                                           ' Выходим из цикла — ничего больше закрывать не нужно
        End If                                                                                                ' Конец условия закрытия сборки

    Loop                                                                                                      ' Повторяем, пока закрываем сборки

End Sub                                                                                                       ' Конец процедуры CloseAsmGroups



' ============================================================================= ' Раздел: ЗАПИСЬ ОДНОЙ СТРОКИ EXCEL
' =============================================================================
Sub WriteExcelRow(ws, rowIndex, depth, startLevel, labelValue, defText, parentPN, qty, codeM, typeN, matO, massEach, massTotal)

    ' Формула в A: для быстрого поиска собираем все G-уровни в одну строку         ' Пояснение
    ws.Cells(rowIndex, COL_A_FIND).Formula = "=CONCATENATE(B" & rowIndex & ",C" & rowIndex & ",D" & rowIndex & ",E" & rowIndex & ",F" & rowIndex & ",G" & rowIndex & ",H" & rowIndex & ",I" & rowIndex & ")" ' A

    ' Перед записью уровня очищаем B..I, чтобы была только одна “ступень”          ' Пояснение
    ws.Range(ws.Cells(rowIndex, COL_G1), ws.Cells(rowIndex, COL_G8)).Value = ""   ' Очищаем уровеньные колонки G1..G8

    ' Рассчитываем колонку для уровня: стартовый G + depth                         ' Пояснение
    Dim levelCol                                                                  ' Колонка, куда писать labelValue
    levelCol = COL_G1 + (CLng(startLevel) - 1) + CLng(depth)                      ' B + (start-1) + depth

    If levelCol < COL_G1 Then levelCol = COL_G1                                   ' Защита от ухода левее B
    If levelCol > COL_G8 Then levelCol = COL_G8                                   ' Защита от ухода правее I (G8)

    ws.Cells(rowIndex, levelCol).Value = labelValue                               ' Пишем PartNumber/крепеж/ПКИ на нужный уровень

    ws.Cells(rowIndex, COL_J_DEF).Value = defText                                 ' J: Definition (Наименование ЭМ)
    ws.Cells(rowIndex, COL_K_PARENT).Value = parentPN                             ' K: Входимость (обозначение родителя)
    ws.Cells(rowIndex, COL_L_QTY).Value = qty                                     ' L: Количество

    ws.Cells(rowIndex, COL_M_CODE).Value = codeM                                  ' M: код
    ws.Cells(rowIndex, COL_N_TYPE).Value = typeN                                  ' N: Композит/металл
    ws.Cells(rowIndex, COL_O_MAT).Value = matO                                    ' O: Материал
    ws.Cells(rowIndex, COL_P_MASS_EACH).Value = massEach                          ' P: Масса 1 шт
    ws.Cells(rowIndex, COL_Q_MASS_TOTAL).Value = massTotal                        ' Q: Суммарная масса

End Sub                                                                            ' Конец WriteExcelRow


' ============================================================================= ' Раздел: ШАПКА ТАБЛИЦЫ (СТРОКА 1)
' =============================================================================
Sub WriteHeader(ws)

    ws.Cells(1, 1).Value = "Найти (ЭЧ)"                                           ' A
    ws.Cells(1, 2).Value = "G1"                                                   ' B
    ws.Cells(1, 3).Value = "G2"                                                   ' C
    ws.Cells(1, 4).Value = "G3"                                                   ' D
    ws.Cells(1, 5).Value = "G4"                                                   ' E
    ws.Cells(1, 6).Value = "G5"                                                   ' F
    ws.Cells(1, 7).Value = "G6"                                                   ' G
    ws.Cells(1, 8).Value = "G7"                                                   ' H
    ws.Cells(1, 9).Value = "G8"                                                   ' I
    ws.Cells(1, 10).Value = "Наименование (ЭМ)"                                  ' J
    ws.Cells(1, 11).Value = "Входимость"                                         ' K
    ws.Cells(1, 12).Value = "Кол-во на 1 сборку"                                 ' L
    ws.Cells(1, 13).Value = "код"                                                ' M
    ws.Cells(1, 14).Value = "Композит/металл"                                    ' N
    ws.Cells(1, 15).Value = "Материал"                                           ' O
    ws.Cells(1, 16).Value = "Масса 1 шт. (кг)"                                   ' P
    ws.Cells(1, 17).Value = "Суммарная масса (кг)"                               ' Q

End Sub                                                                            ' Конец WriteHeader


' ============================================================================= ' Раздел: ФОРМАТИРОВАНИЕ + OUTLINE
' =============================================================================
Sub FormatSheet(ws, lastRow)

    ' -----------------------------                                                ' Ширины столбцов (примерно как на твоих скринах)
    ws.Columns("A").ColumnWidth = 24                                              ' Ширина A
    ws.Columns("B").ColumnWidth = 3                                               ' Ширина B
    ws.Columns("C").ColumnWidth = 4                                               ' Ширина C
    ws.Columns("D").ColumnWidth = 8                                               ' Ширина D
    ws.Columns("E").ColumnWidth = 8                                               ' Ширина E
    ws.Columns("F").ColumnWidth = 8                                               ' Ширина F
    ws.Columns("G").ColumnWidth = 7                                               ' Ширина G
    ws.Columns("H").ColumnWidth = 5                                               ' Ширина H
    ws.Columns("I").ColumnWidth = 9                                               ' Ширина I
    ws.Columns("J").ColumnWidth = 45                                              ' Ширина J
    ws.Columns("K").ColumnWidth = 23                                              ' Ширина K
    ws.Columns("L").ColumnWidth = 12                                              ' Ширина L
    ws.Columns("M").ColumnWidth = 10                                              ' Ширина M
    ws.Columns("N").ColumnWidth = 14                                              ' Ширина N
    ws.Columns("O").ColumnWidth = 40                                              ' Ширина O
    ws.Columns("P").ColumnWidth = 12                                              ' Ширина P
    ws.Columns("Q").ColumnWidth = 14                                              ' Ширина Q

    ' -----------------------------                                                ' Перенос текста в заголовках и оформление шапки
    Dim hdr                                                                       ' Диапазон шапки A1:Q1
    Set hdr = ws.Range(ws.Cells(1, 1), ws.Cells(1, COL_Q_MASS_TOTAL))             ' Берём диапазон шапки

    hdr.Font.Bold = True                                                         ' Делаем шапку жирной
    hdr.HorizontalAlignment = -4108                                               ' Выравнивание по центру (xlCenter = -4108)
    hdr.VerticalAlignment = -4108                                                 ' Выравнивание по центру по вертикали
    hdr.WrapText = True                                                           ' Включаем перенос текста в заголовках
    hdr.Interior.Color = RGBLong(221, 235, 247)                                   ' Светло-синяя заливка (как у Excel "Light Blue")
    ws.Rows(1).RowHeight = 30                                                     ' Чуть увеличиваем высоту строки 1 (чтобы перенос выглядел нормально)

    ' -----------------------------                                                ' Перенос текста в полезных длинных колонках
    ws.Columns("J").WrapText = True                                               ' Перенос для "Наименование (ЭМ)"
    ws.Columns("O").WrapText = True                                               ' Перенос для "Материал"

    ' -----------------------------                                                ' Границы таблицы
    ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, COL_Q_MASS_TOTAL)).Borders.LineStyle = 1 ' Включаем границы (xlContinuous = 1)
    ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, COL_Q_MASS_TOTAL)).Borders.Weight = 2   ' Толщина линий

    ' -----------------------------                                                ' Freeze Panes по строке 2 (шапка остаётся сверху)
    ws.Range("A2").Select                                                         ' Ставим курсор на A2
    ws.Application.ActiveWindow.FreezePanes = True                                 ' Замораживаем области

    ' -----------------------------                                                ' OUTLINE: делаем “кнопки” напротив сборок сверху
    ' В Excel: SummaryRow = xlSummaryAbove => значение 0                            ' Важно: иначе +/- будет внизу группы
    ws.Outline.SummaryRow = 0                                                     ' Итоговая строка (сборка) сверху, а не снизу

    ' -----------------------------                                                ' OUTLINE по столбцам: одна кнопка для скрытия B:I
    On Error Resume Next                                                           ' Перехват (если Excel ругнётся при повторной группировке)
    ws.Columns("B:I").Group                                                        ' Группируем столбцы G1..G8 (получится одна кнопка +/-)
    On Error GoTo 0                                                                ' Возвращаем стандартную обработку ошибок

    ' -----------------------------                                                ' Показываем уровни (чтобы кнопки были видны)
    ws.Outline.ShowLevels 8, 8                                                    ' Показать максимально (строки/столбцы)

End Sub                                                                            ' Конец FormatSheet


' ============================================================================= ' Раздел: UNICODE-ЧТЕНИЕ PAYLOAD (UTF-16 LE)
' =============================================================================
Function ReadUnicodeFile(path)

    On Error Resume Next                                                          ' Перехват ошибок чтения файла

    Dim stm                                                                       ' ADODB.Stream для чтения текста
    Set stm = CreateObject("ADODB.Stream")                                        ' Создаём поток

    stm.Type = 2                                                                  ' Тип = текст
    stm.Charset = "unicode"                                                       ' Кодировка = UTF-16 LE
    stm.Open                                                                      ' Открываем поток
    stm.LoadFromFile path                                                         ' Загружаем файл
    ReadUnicodeFile = stm.ReadText(-1)                                            ' Читаем весь текст
    stm.Close                                                                     ' Закрываем поток

    If Err.Number <> 0 Then                                                       ' Если произошла ошибка
        Err.Clear                                                                 ' Сбрасываем ошибку
        ReadUnicodeFile = ""                                                      ' Возвращаем пустую строку
    End If                                                                        ' Конец обработки ошибок

    On Error GoTo 0                                                               ' Возвращаем стандартную обработку ошибок

End Function                                                                      ' Конец ReadUnicodeFile


' ============================================================================= ' Раздел: ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ
' =============================================================================
Function SafeField(fields, idx)

    If IsArray(fields) = False Then                                               ' Если это не массив
        SafeField = ""                                                            ' Возвращаем пусто
        Exit Function                                                             ' Выходим
    End If                                                                        ' Конец проверки массива

    If idx < 0 Then                                                               ' Если индекс отрицательный
        SafeField = ""                                                            ' Возвращаем пусто
        Exit Function                                                             ' Выходим
    End If                                                                        ' Конец проверки idx

    If idx > UBound(fields) Then                                                  ' Если индекс за пределами массива
        SafeField = ""                                                            ' Возвращаем пусто
        Exit Function                                                             ' Выходим
    End If                                                                        ' Конец проверки UBound

    SafeField = CStr(fields(idx))                                                 ' Возвращаем поле как строку

End Function                                                                      ' Конец SafeField


Function ParseLocaleNumber(s)

    Dim t                                                                         ' Рабочая строка
    t = Trim(CStr(s))                                                             ' Приводим к строке и обрезаем пробелы

    If t = "" Then                                                                ' Если пусто
        ParseLocaleNumber = 0                                                     ' Возвращаем 0
        Exit Function                                                             ' Выходим
    End If                                                                        ' Конец проверки пустоты

    Dim decSep                                                                    ' Десятичный разделитель системы ('.' или ',')
    decSep = Mid(CStr(1/2), 2, 1)                                                 ' Достаём системный разделитель

    t = Replace(t, ".", decSep)                                                   ' Заменяем '.' на системный
    t = Replace(t, ",", decSep)                                                   ' Заменяем ',' на системный

    On Error Resume Next                                                          ' Перехват ошибок CDbl
    ParseLocaleNumber = CDbl(t)                                                   ' Преобразуем в Double
    If Err.Number <> 0 Then                                                       ' Если преобразование не удалось
        Err.Clear                                                                 ' Сбрасываем ошибку
        ParseLocaleNumber = 0                                                     ' Возвращаем 0
    End If                                                                        ' Конец проверки Err
    On Error GoTo 0                                                               ' Возвращаем стандартную обработку ошибок

End Function                                                                      ' Конец ParseLocaleNumber


Function ParseLong(s)

    Dim d                                                                         ' Double-значение (может быть с дробной частью)
    d = ParseLocaleNumber(s)                                                      ' Парсим как Double с учётом локали

    ParseLong = CLng(Fix(d))                                                      ' Fix() отбрасывает дробную часть (без округления)

End Function                                                                      ' Конец ParseLong


Function RGBLong(r, g, b)

    ' Excel ожидает число как r + g*256 + b*65536 (как VBA RGB)                    ' Пояснение формулы
    RGBLong = CLng(r) + CLng(g) * 256 + CLng(b) * 65536                           ' Возвращаем Long-цвет

End Function                                                                      ' Конец RGBLong
